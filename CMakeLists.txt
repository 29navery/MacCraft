cmake_minimum_required(VERSION 3.20)
project(MacCraft VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(APPLE AND NOT CMAKE_OSX_ARCHITECTURES)
    set(CMAKE_OSX_ARCHITECTURES arm64 CACHE STRING "MacCraft native architecture" FORCE)
endif()
add_executable(maccraft-graphics-probe native/graphics_probe.cpp)
target_link_libraries(maccraft-graphics-probe PRIVATE ${CMAKE_DL_LIBS})
target_compile_options(maccraft-graphics-probe PRIVATE -Wall -Wextra -Wpedantic)
install(TARGETS maccraft-graphics-probe RUNTIME DESTINATION bin)
include(CTest)
if(BUILD_TESTING)
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    add_test(NAME package-tests COMMAND ${Python3_EXECUTABLE} -m unittest discover -s tests -v)
    set_tests_properties(package-tests PROPERTIES WORKING_DIRECTORY ${CMAKE_SOURCE_DIR})
    add_test(NAME probe-rejects-missing-driver COMMAND maccraft-graphics-probe /nonexistent/angle)
    set_tests_properties(probe-rejects-missing-driver PROPERTIES WILL_FAIL TRUE)
endif()
